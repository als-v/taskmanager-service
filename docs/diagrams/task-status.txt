sequenceDiagram
    actor ACT as MEMBER/ADMIN
    participant TC as TaskController
    participant TS as TaskService
    participant POL as ProjectAccessPolicy
    participant TR as TaskRepository
    participant ALR as AuditLogRepository

    ACT->>TC: PATCH /api/projects/{id}/tasks/{taskId} { status = NOVO }
    TC->>TS: atualizarStatus(projectId, taskId, novoStatus, actorId)
    TS->>POL: garantirMembroDoProjeto(actorId, projectId)
    TS->>TR: findById(taskId)
    TR-->>TS: task (statusAtual, priority, assigneeId, taskProjectId)
    TS->>TS: validar taskProjectId == projectId

    alt task NÃO pertence ao projeto da rota
        TS-->>TC: throws 404
        TC-->>ACT: 404
    else task pertence ao projeto
        alt statusAtual = DONE e novoStatus = TODO
            TS-->>TC: throws 422 (DONE não volta a TODO)
            TC-->>ACT: 422
        else novoStatus = DONE e priority = CRITICAL
            TS->>POL: garantirAdminDoProjeto(actorId, projectId)
            alt actor NÃO é ADMIN do projeto
                TS-->>TC: throws 422 (CRITICAL só ADMIN fecha)
                TC-->>ACT: 422
            end
        end

        opt novoStatus = IN_PROGRESS
            TS->>TR: count where projectId = ? and assigneeId = ? and status = IN_PROGRESS
            TR-->>TS: total
            alt total >= 5 (WIP limit por projeto)
                TS-->>TC: throws 422 (limite de 5 IN_PROGRESS por responsável)
                TC-->>ACT: 422
            end
        end

        TS->>TR: save(task com novoStatus)
        TS->>ALR: save(action = STATUS_CHANGED, from = statusAtual, to = novoStatus)
        TS-->>TC: TaskResponse
        TC-->>ACT: 200 OK
    end
