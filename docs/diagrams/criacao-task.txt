sequenceDiagram
    actor ACT as MEMBER/ADMIN
    participant TC as TaskController
    participant TS as TaskService
    participant POL as ProjectAccessPolicy
    participant TR as TaskRepository
    participant ALR as AuditLogRepository
    participant NS as NotificationService
    participant NR as NotificationRepository
    participant UNR as UserNotificationRepository

    ACT->>TC: POST /api/projects/{id}/tasks { title, ..., assigneeId? }
    TC->>TS: criar(projectId, dto, actorId)
    TS->>POL: garantirMembroDoProjeto(actorId, projectId)
    TS->>TR: save(task)
    TS->>ALR: save(action = TASK_CREATED, actor, task)

    alt assigneeId informado (não nulo)
        TS->>POL: garantirMembroDoProjeto(assigneeId, projectId)
        TS->>ALR: save(action = ASSIGNEE_CHANGED, task)
        TS->>NS: notificar(assigneeId, TASK_ASSIGNED, projectId, taskId)
        NS->>NR: save(notification)
        NR-->>NS: notification (id)
        NS->>UNR: save(user_notification { assigneeId, notificationId })
    else sem responsável
        Note over TS: task criada sem usuario, logo nenhuma notificação e auditoria
    end

    TS-->>TC: TaskResponse
    TC-->>ACT: 201 Created
